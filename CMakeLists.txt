cmake_minimum_required(VERSION 3.16)
project(LMS VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

# AUTO preserves optional MySQL; ON fails configuration if the client is missing.
set(LMS_MYSQL "AUTO" CACHE STRING "MySQL support: AUTO, ON, OFF")
set_property(CACHE LMS_MYSQL PROPERTY STRINGS AUTO ON OFF)
if(NOT LMS_MYSQL MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "LMS_MYSQL must be AUTO, ON or OFF")
endif()
set(LMS_CORE_SOURCES src/user_repo_file.cpp src/order_repo_file.cpp src/user_service.cpp)
if(NOT LMS_MYSQL STREQUAL "OFF")
  find_path(MYSQL_INCLUDE NAMES mysql.h
    HINTS "${MYSQL_ROOT}" ENV MYSQL_ROOT
    PATHS "E:/Program Files/MySQL/MySQL Server 8.0"
    PATH_SUFFIXES include include/mysql include/mariadb mysql mariadb)
  find_library(MYSQL_LIB NAMES mysqlclient mariadb libmysql
    HINTS "${MYSQL_ROOT}" ENV MYSQL_ROOT
    PATHS "E:/Program Files/MySQL/MySQL Server 8.0"
    PATH_SUFFIXES lib lib64)
  if(MYSQL_INCLUDE AND MYSQL_LIB)
    list(APPEND LMS_CORE_SOURCES src/user_repo_mysql.cpp)
    set(LMS_HAVE_MYSQL TRUE)
  elseif(LMS_MYSQL STREQUAL "ON")
    message(FATAL_ERROR "MySQL client not found. Set MYSQL_ROOT or MYSQL_INCLUDE and MYSQL_LIB.")
  endif()
endif()
add_library(lms-core STATIC ${LMS_CORE_SOURCES})
target_include_directories(lms-core PUBLIC include)
target_link_libraries(lms-core PUBLIC Threads::Threads)
if(LMS_HAVE_MYSQL)
  target_compile_definitions(lms-core PUBLIC LMS_HAS_MYSQL)
  target_include_directories(lms-core PUBLIC "${MYSQL_INCLUDE}")
  target_link_libraries(lms-core PUBLIC "${MYSQL_LIB}")
  add_executable(mysql-pool-test tests/mysql_pool.cpp)
  target_link_libraries(mysql-pool-test PRIVATE lms-core)
  message(STATUS "MySQL support enabled")
else()
  message(STATUS "File-only build; requesting MySQL at runtime will fail explicitly")
endif()
add_executable(lms-storage src/storage_tool.cpp)
target_include_directories(lms-storage PRIVATE include)
target_link_libraries(lms-storage PRIVATE Threads::Threads)
add_executable(lms-server src/main.cpp)
target_link_libraries(lms-server PRIVATE lms-core)
if(MSVC)
  target_compile_options(lms-core PRIVATE /W3 /utf-8)
  target_compile_options(lms-server PRIVATE /W3 /utf-8)
  target_compile_options(lms-storage PRIVATE /W3 /utf-8)
endif()
if(WIN32)
  target_link_libraries(lms-server PRIVATE ws2_32)
endif()
# Keep config and static page available when launching from the build directory.
file(COPY config static DESTINATION "${CMAKE_BINARY_DIR}")

enable_testing()
option(BUILD_HTTP_TESTS "Build lifecycle and real HTTP regression tests" ON)
if(BUILD_HTTP_TESTS)
  add_executable(server-lifecycle tests/server_lifecycle.cpp)
  target_include_directories(server-lifecycle PRIVATE include)
  target_link_libraries(server-lifecycle PRIVATE Threads::Threads)
  if(WIN32)
    target_link_libraries(server-lifecycle PRIVATE ws2_32)
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/tmp")
  foreach(case IN ITEMS stage1_http startup_config)
    add_test(NAME ${case} COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/${case}.py $<TARGET_FILE:lms-server>)
    set_tests_properties(${case} PROPERTIES ENVIRONMENT
      "TEMP=${CMAKE_BINARY_DIR}/tmp;TMP=${CMAKE_BINARY_DIR}/tmp;TMPDIR=${CMAKE_BINARY_DIR}/tmp")
    if(NOT LMS_HAVE_MYSQL)
      set_property(TEST ${case} APPEND PROPERTY ENVIRONMENT "LMS_TEST_FILE_ONLY=1")
    endif()
  endforeach()
  add_test(NAME storage_http COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/storage_http.py $<TARGET_FILE:lms-server> $<TARGET_FILE:lms-storage>)
  add_test(NAME lifecycle COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_lifecycle.py $<TARGET_FILE:server-lifecycle>)
endif()
# Preserve the existing optional GoogleTest target. No download unless requested.
option(BUILD_TESTS "Build GoogleTest unit tests" OFF)
if(BUILD_TESTS)
  include(FetchContent)
  FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.14.0)
  FetchContent_MakeAvailable(googletest)
  add_executable(lms-tests tests/test_json.cpp tests/test_crypto.cpp tests/test_jwt.cpp tests/test_service.cpp)
  target_link_libraries(lms-tests PRIVATE lms-core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(lms-tests)
endif()
